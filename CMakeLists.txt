cmake_minimum_required(VERSION 3.22)
project(SubtitleDesk VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Multimedia)
option(SUBTITLEDESK_BUILD_TESTS "Build document, decoder and GUI tests" ON)
set(SUBTITLEDESK_RUNTIME_ROOT "" CACHE PATH "Extracted, verified runtime for media and recognition tests")
set(SUBTITLEDESK_TEST_MEDIA "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/speech.wav" CACHE FILEPATH "16 kHz mono PCM speech for recognition tests")
set(SUBTITLEDESK_TEST_IMPORT_MODEL "" CACHE FILEPATH "Optional compatible external Whisper model for real import tests")
set(APP_SOURCES src/mainwindow.cpp src/mainwindow.h src/model_manager_dialog.cpp src/model_manager_dialog.h src/audio_session.cpp src/audio_session.h src/transcription_service.cpp src/transcription_service.h src/waveform.cpp src/waveform.h)
add_library(subtitle_document STATIC src/subtitle_document.cpp src/subtitle_document.h)
target_include_directories(subtitle_document PUBLIC src)
target_link_libraries(subtitle_document PUBLIC Qt6::Core)
add_library(model_catalog STATIC src/model_catalog.cpp src/model_catalog.h)
target_include_directories(model_catalog PUBLIC src)
target_link_libraries(model_catalog PUBLIC Qt6::Core)
add_library(subtitle_ui STATIC ${APP_SOURCES})
target_include_directories(subtitle_ui PUBLIC src PRIVATE vendor/ffmpeg/include)
target_link_libraries(subtitle_ui PUBLIC subtitle_document model_catalog Qt6::Widgets Qt6::Multimedia)
add_executable(SubtitleDesk WIN32 src/main.cpp resources/app.qrc)
target_link_libraries(SubtitleDesk PRIVATE subtitle_ui)
if(WIN32)
    enable_language(RC)
    target_sources(SubtitleDesk PRIVATE resources/windows.rc)
    target_include_directories(SubtitleDesk PRIVATE resources)
else()
    set_target_properties(SubtitleDesk PROPERTIES OUTPUT_NAME subtitledesk BUILD_WITH_INSTALL_RPATH TRUE BUILD_RPATH "$ORIGIN/lib;$ORIGIN/../lib/subtitledesk/lib" INSTALL_RPATH "$ORIGIN/lib;$ORIGIN/../lib/subtitledesk/lib")
endif()
if(SUBTITLEDESK_BUILD_TESTS)
    enable_testing()
    find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
    add_executable(test_document tests/test_document.cpp)
    target_link_libraries(test_document PRIVATE subtitle_document)
    add_test(NAME document_formats COMMAND test_document)
    set_tests_properties(document_formats PROPERTIES TIMEOUT 30)
    add_test(NAME packaging_safety COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p "test_packaging.py" -v)
    set_tests_properties(packaging_safety PROPERTIES TIMEOUT 30)
    add_executable(test_transcription tests/test_transcription.cpp src/transcription_service.cpp src/transcription_service.h)
    target_link_libraries(test_transcription PRIVATE subtitle_document model_catalog Qt6::Core)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(test_model_catalog tests/test_model_catalog.cpp)
    target_link_libraries(test_model_catalog PRIVATE model_catalog Qt6::Test)
    add_test(NAME model_catalog COMMAND test_model_catalog)
    set_tests_properties(model_catalog PROPERTIES TIMEOUT 300 RESOURCE_LOCK speech_models)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_audio.cpp")
        add_executable(test_audio tests/test_audio.cpp src/audio_session.cpp src/audio_session.h)
        target_include_directories(test_audio PRIVATE src vendor/ffmpeg/include)
        target_link_libraries(test_audio PRIVATE Qt6::Core Qt6::Multimedia)
    endif()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ui.cpp")
        find_package(Qt6 REQUIRED COMPONENTS Test)
        add_executable(test_ui tests/test_ui.cpp)
        target_link_libraries(test_ui PRIVATE subtitle_ui Qt6::Test)
        add_test(NAME editor_workflow COMMAND test_ui)
        set_tests_properties(editor_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_MEDIA_BACKEND=ffmpeg")
        set_tests_properties(editor_workflow PROPERTIES TIMEOUT 240 RESOURCE_LOCK speech_models)
    endif()

    # The application is offline. Only this explicit build configuration enables
    # tests that need downloaded runtimes; a normal source build never fetches them.
    if(SUBTITLEDESK_RUNTIME_ROOT)
        get_filename_component(SUBTITLEDESK_RUNTIME_ROOT "${SUBTITLEDESK_RUNTIME_ROOT}" ABSOLUTE)
        if(NOT EXISTS "${SUBTITLEDESK_RUNTIME_ROOT}/resources/models/whistle.cact")
            message(FATAL_ERROR "SUBTITLEDESK_RUNTIME_ROOT must contain the extracted, verified runtime.")
        endif()
        if(NOT EXISTS "${SUBTITLEDESK_TEST_MEDIA}")
            message(FATAL_ERROR "SUBTITLEDESK_TEST_MEDIA must be an existing 16 kHz mono PCM speech WAV.")
        endif()
        find_program(SUBTITLEDESK_FFMPEG NAMES ffmpeg REQUIRED)
        find_program(SUBTITLEDESK_FFPROBE NAMES ffprobe REQUIRED)
        set(AUDIO_FIXTURES "${CMAKE_CURRENT_BINARY_DIR}/audio-fixtures")
        add_test(NAME create_audio_fixtures COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/create_audio_fixtures.py" "${AUDIO_FIXTURES}"
            --ffmpeg "${SUBTITLEDESK_FFMPEG}" --ffprobe "${SUBTITLEDESK_FFPROBE}")
        set_tests_properties(create_audio_fixtures PROPERTIES FIXTURES_SETUP audio_fixtures TIMEOUT 120)
        add_test(NAME audio_decoding COMMAND test_audio "${AUDIO_FIXTURES}")
        set_tests_properties(audio_decoding PROPERTIES FIXTURES_REQUIRED audio_fixtures TIMEOUT 180)
        set(RUNTIME_TESTS audio_decoding editor_workflow)
        foreach(MODEL whistle tiny base small)
            add_test(NAME transcription_${MODEL} COMMAND test_transcription
                "${SUBTITLEDESK_RUNTIME_ROOT}/resources" "${SUBTITLEDESK_TEST_MEDIA}" "${MODEL}")
            set_tests_properties(transcription_${MODEL} PROPERTIES TIMEOUT 240 RESOURCE_LOCK speech_models)
            list(APPEND RUNTIME_TESTS transcription_${MODEL})
        endforeach()
        add_test(NAME transcription_cancel COMMAND test_transcription
            "${SUBTITLEDESK_RUNTIME_ROOT}/resources" "${SUBTITLEDESK_TEST_MEDIA}" small --cancel)
        set_tests_properties(transcription_cancel PROPERTIES TIMEOUT 30 RESOURCE_LOCK speech_models)
        list(APPEND RUNTIME_TESTS transcription_cancel)
        add_test(NAME transcription_imported_whistle COMMAND test_transcription
            "${SUBTITLEDESK_RUNTIME_ROOT}/resources" "${SUBTITLEDESK_TEST_MEDIA}" imported
            --import-model "${SUBTITLEDESK_RUNTIME_ROOT}/resources/models/whistle.cact")
        set_tests_properties(transcription_imported_whistle PROPERTIES TIMEOUT 600 RESOURCE_LOCK speech_models)
        list(APPEND RUNTIME_TESTS transcription_imported_whistle)
        set_property(TEST editor_workflow APPEND PROPERTY ENVIRONMENT
            "SUBTITLEDESK_RESOURCES=${SUBTITLEDESK_RUNTIME_ROOT}/resources"
            "SUBTITLEDESK_TEST_MEDIA=${SUBTITLEDESK_TEST_MEDIA}")
        set_property(TEST model_catalog APPEND PROPERTY ENVIRONMENT
            "SUBTITLEDESK_RESOURCES=${SUBTITLEDESK_RUNTIME_ROOT}/resources")
        list(APPEND RUNTIME_TESTS model_catalog)
        if(SUBTITLEDESK_TEST_IMPORT_MODEL)
            if(NOT EXISTS "${SUBTITLEDESK_TEST_IMPORT_MODEL}")
                message(FATAL_ERROR "SUBTITLEDESK_TEST_IMPORT_MODEL must be an existing compatible model file.")
            endif()
            set_property(TEST model_catalog editor_workflow APPEND PROPERTY ENVIRONMENT
                "SUBTITLEDESK_TEST_IMPORT_MODEL=${SUBTITLEDESK_TEST_IMPORT_MODEL}")
            add_test(NAME transcription_imported COMMAND test_transcription
                "${SUBTITLEDESK_RUNTIME_ROOT}/resources" "${SUBTITLEDESK_TEST_MEDIA}" imported
                --import-model "${SUBTITLEDESK_TEST_IMPORT_MODEL}")
            add_test(NAME transcription_imported_cancel COMMAND test_transcription
                "${SUBTITLEDESK_RUNTIME_ROOT}/resources" "${SUBTITLEDESK_TEST_MEDIA}" imported
                --import-model "${SUBTITLEDESK_TEST_IMPORT_MODEL}" --cancel)
            set_tests_properties(transcription_imported transcription_imported_cancel PROPERTIES
                TIMEOUT 600 RESOURCE_LOCK speech_models)
            list(APPEND RUNTIME_TESTS transcription_imported transcription_imported_cancel)
        endif()
        if(WIN32)
            # Resolve the pinned FFmpeg DLLs while retaining the SDK/MinGW paths.
            set_property(TEST ${RUNTIME_TESTS} APPEND PROPERTY ENVIRONMENT_MODIFICATION
                "PATH=path_list_prepend:${SUBTITLEDESK_RUNTIME_ROOT}")
        else()
            set_property(TEST ${RUNTIME_TESTS} APPEND PROPERTY ENVIRONMENT_MODIFICATION
                "LD_LIBRARY_PATH=path_list_prepend:${SUBTITLEDESK_RUNTIME_ROOT}/lib")
        endif()
        add_test(NAME packaged_runtime COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/packaging/validate_package.py" --platform
            $<IF:$<BOOL:${WIN32}>,windows,linux> --runtime-only --path "${SUBTITLEDESK_RUNTIME_ROOT}")
        set_tests_properties(packaged_runtime PROPERTIES TIMEOUT 120)
    endif()
endif()
